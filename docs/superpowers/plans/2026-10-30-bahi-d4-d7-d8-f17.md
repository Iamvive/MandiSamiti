```kotlin
/** D4.3/D4.8: simple interest on the unpaid principal of each period, minus interest already received. Never negative. */
fun interestDue(principalPaisa: Long, monthlyRatePercent: Double, lentAtMs: Long, settlements: List<LendSettlement>, nowMs: Long): Long {
    var outstanding = principalPaisa
    var from = lentAtMs
    var accrued = 0L
    for (s in settlements.filter { !it.isVoid }.sortedBy { it.settledAt }) {
        accrued += calculateAccruedInterestBetweenEpochs(outstanding, monthlyRatePercent, from, s.settledAt).accruedInterestPaisa
        outstanding = (outstanding - s.principalPaisa).coerceAtLeast(0L)
        from = s.settledAt
    }
    accrued += calculateAccruedInterestBetweenEpochs(outstanding, monthlyRatePercent, from, nowMs).accruedInterestPaisa
    val paid = settlements.filter { !it.isVoid }.sumOf { it.interestPaisa }
    return (accrued - paid).coerceAtLeast(0L)
}
```
